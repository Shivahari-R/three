Installation

Project structure

Every three.js project needs at least one HTML file to define the webpage, and a JavaScript file to run your three.js code. The structure and naming choices below aren't required, but will be used throughout this guide for consistency.

  • index.html
<!DOCTYPE html>
<html lang="en">
  <head>
    <meta charset="utf-8">
    <title>My first three.js app</title>
    <style>
      body { margin: 0; }

      #crosshair {
        position: fixed;
        top: 50%;
        left: 50%;
        transform: translate(-50%, -50%);
        color: white;
        font-size: 25px;
        pointer-events: none;
      }
    </style>
  </head>
  <body>
    <!-- Start Screen -->
    <div id="startScreen">
      <h4>Shooting Game</h4>
      <button onclick="startGame()">START</button>
    </div>

    <!-- Game Screen -->
    <div id="gameScreen" style="display: none;">
      <div id="score">Score: 0</div>
      <button onclick="restart()">Restart</button>
      <div id="crosshair">+</div>
    </div>

    <script type="module" src="/main.js"></script>
  </body>
</html>

Next, create a JavaScript file called main.js in the same folder.

  • main.js
import * as THREE from 'three';

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera( 75, window.innerWidth / window.innerHeight, 0.1, 1000 );

const renderer = new THREE.WebGLRenderer();
renderer.setSize( window.innerWidth, window.innerHeight );
document.body.appendChild( renderer.domElement );

Install and run

Open a terminal in your project folder and install three.js and Vite, a development server that lets you run your project locally.

# three.js
npm install --save three

# vite
npm install --save-dev vite
Installation added node_modules/ and package.json to my project. What are they?

These files and folders are created by npm to manage the packages used by your project.

Using three.js with TypeScript

Three.js can also be used in TypeScript projects.

  1. From your terminal, run:
npx vite
What is npx?

npx runs packages from the npm ecosystem without requiring a global installation.

  1. If everything went well, you'll see a URL like http://localhost:5173 appear in your terminal, and can open that URL to see your web application.

The page will be blank — you're ready to create a scene.

If you want to learn more about these tools before you continue, see:

            player.material.color.setHex(0xffffff); // Turn player WHITE to prove it works

        }

}

import * as THREE from 'three';

// SCENE
const scene = new THREE.Scene();
scene.background = new THREE.Color("lightblue");

// CAMERA
const camera = new THREE.PerspectiveCamera(
    75,
    window.innerWidth / window.innerHeight,
    0.1,
    1000
);
camera.position.z = 10;
camera.lookAt(0, 0, 0);

// RENDERER
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// PLATFORM
const platformGeometry = new THREE.BoxGeometry(12, 0.5, 2);
const platformMaterial = new THREE.MeshBasicMaterial({ color: "grey" });
const platform = new THREE.Mesh(platformGeometry, platformMaterial);
platform.position.y = -3;
scene.add(platform);

// BUCKET
const bucketGeometry = new THREE.BoxGeometry(1.5, 0.5, 1);
const bucketMaterial = new THREE.MeshBasicMaterial({ color: "blue" });
const bucket = new THREE.Mesh(bucketGeometry, bucketMaterial);
bucket.position.y = -2.5;
scene.add(bucket);

// COLLISION SETUP - initialized once for performance
const bucketBox = new THREE.Box3();
const sphereBox = new THREE.Box3();
const platformBox = new THREE.Box3();

// VARIABLES
let score = 0;
let velocity = 0;
let gravity = 0.001;
let gameRunning = false;
let sphere = null;

// MEMORY LEAK FIX: Create geometry and material once to reuse
const sphereGeometry = new THREE.SphereGeometry(0.3, 16, 16);
const sphereMaterial = new THREE.MeshBasicMaterial({ color: "red" });

// CREATE SPHERE
function createSphere() {
    sphere = new THREE.Mesh(sphereGeometry, sphereMaterial);
    sphere.position.set(Math.random() * 10 - 5, 4, 0);
    velocity = 0;
    scene.add(sphere);
}

// START SCREEN
const start = document.querySelector(".start");
const btn = document.getElementById("startbtn");

renderer.domElement.style.display = "none";

btn.addEventListener("click", function() {
    start.style.display = "none";
    renderer.domElement.style.display = "block";

    score = 0;
    gameRunning = true;
    bucket.position.x = 0;

    document.getElementById("score").innerText = "Score: 0";
    document.getElementById("message").innerText = "";

    if (sphere) {
        scene.remove(sphere);
    }
    createSphere();
});

// BUCKET MOVEMENT
window.addEventListener("keydown", function(event) {
    if (!gameRunning) return;

    if (event.key === "ArrowLeft") {
        bucket.position.x -= 0.3;
    }

    if (event.key === "ArrowRight") {
        bucket.position.x += 0.3;
    }

    bucket.position.x = THREE.MathUtils.clamp(bucket.position.x, -5, 5);
});

// ANIMATION
function animate() {
    if (gameRunning && sphere) {
        // GRAVITY
        velocity += gravity;
        sphere.position.y -= velocity;

        // UPDATE BOUNDING BOXES
        bucketBox.setFromObject(bucket);
        sphereBox.setFromObject(sphere);
        platformBox.setFromObject(platform);

        // BUCKET COLLISION
        if (bucketBox.intersectsBox(sphereBox)) {
            scene.remove(sphere);
            sphere = null;

            score++;
            document.getElementById("score").innerText = "Score: " + score;

            createSphere();
        }
        // GROUND COLLISION (Using Bounding Box)
        else if (platformBox.intersectsBox(sphereBox)) {
            scene.remove(sphere);
            sphere = null;

            gameRunning = false;
            document.getElementById("message").innerText = "Missed! Game Over";
            start.style.display = "block"; // Show start screen again to retry
        }
    }

    renderer.render(scene, camera);
}

// START ANIMATION
renderer.setAnimationLoop(animate);

Coin Game

import * as THREE from 'three';

// 1. SCENE SETUP
const scene = new THREE.Scene();
scene.background = new THREE.Color("skyblue");

// Angle the camera down slightly to see depth (Z-axis)
const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.set(0, 8, 15);
camera.lookAt(0, 0, 0);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// 2. LARGE PLATFORM (Width: 20, Height: 1, Depth: 20)
const platform = new THREE.Mesh(
  new THREE.BoxGeometry(20, 1, 20),
  new THREE.MeshBasicMaterial({ color: "grey" })
);
platform.position.y = -0.5; // Top surface is exactly at Y = 0
scene.add(platform);

// 3. PLAYER CUBE
const player = new THREE.Mesh(
  new THREE.BoxGeometry(1, 1, 1),
  new THREE.MeshBasicMaterial({ color: "red" })
);
player.position.set(0, 0.5, 0); // Resting exactly on top of the platform
scene.add(player);

// 4. MULTIPLE COINS
const coins = [];
const coinGeo = new THREE.BoxGeometry(0.5, 0.5, 0.5);
const coinMat = new THREE.MeshBasicMaterial({ color: "yellow" });

// Spawn 10 coins in random locations across the platform
for (let i = 0; i < 10; i++) {
  const coin = new THREE.Mesh(coinGeo, coinMat);
  // Random X and Z between -9 and 9
  const randomX = Math.random() * 18 - 9;
  const randomZ = Math.random() * 18 - 9;
  
  coin.position.set(randomX, 0.5, randomZ);
  scene.add(coin);
  coins.push(coin);
}

// 5. GAME VARIABLES & PHYSICS
const keys = {};
let yVelocity = 0;
let isJumping = false;
const gravity = -0.015;
let score = 0;
let gameRunning = true;

const playerBox = new THREE.Box3();
const coinBox = new THREE.Box3();

// 6. CONTROLS (X, Z, and Space for Jump)
window.addEventListener("keydown", (e) => {
  keys[e.code] = true;
  
  // Jump on Spacebar
  if (e.code === "Space" && !isJumping && gameRunning) {
    yVelocity = 0.35; 
    isJumping = true;
  }
});
window.addEventListener("keyup", (e) => keys[e.code] = false);

// 7. ANIMATION LOOP
function animate() {
  if (gameRunning) {
    
    // --- MOVEMENT (X and Z Axes) ---
    if (keys["ArrowUp"]) player.position.z -= 0.15;    // Move Forward (Negative Z)
    if (keys["ArrowDown"]) player.position.z += 0.15;  // Move Backward (Positive Z)
    if (keys["ArrowLeft"]) player.position.x -= 0.15;  // Move Left (Negative X)
    if (keys["ArrowRight"]) player.position.x += 0.15; // Move Right (Positive X)

    // --- JUMPING / GRAVITY (Y Axis) ---
    yVelocity += gravity;
    player.position.y += yVelocity;

    // --- PLATFORM COLLISION ---
    // Check if player is within the platform's X and Z bounds
    const onPlatformX = player.position.x > -10.5 && player.position.x < 10.5;
    const onPlatformZ = player.position.z > -10.5 && player.position.z < 10.5;

    // If over the platform and hitting the surface (Y <= 0.5)
    if (onPlatformX && onPlatformZ && player.position.y <= 0.5) {
      player.position.y = 0.5; // Snap to surface
      yVelocity = 0;           // Stop falling
      isJumping = false;       // Allow jumping again
    }

    // --- FALLING / GAME OVER ---
    if (player.position.y < -5) {
      gameRunning = false;
      document.getElementById("gameOver").style.display = "block";
    }

    // --- COIN COLLECTION ---
    playerBox.setFromObject(player);

    // Loop backwards through the array to safely remove collected coins
    for (let i = coins.length - 1; i >= 0; i--) {
      const currentCoin = coins[i];
      
      // Spin the coins
      currentCoin.rotation.y += 0.05;
      currentCoin.rotation.x += 0.05;

      coinBox.setFromObject(currentCoin);

      // Check collision
      if (playerBox.intersectsBox(coinBox)) {
        scene.remove(currentCoin); // Remove from 3D scene
        coins.splice(i, 1);        // Remove from our tracking array
        
        score++;
        document.getElementById("score").innerText = "Score: " + score;
      }
    }
  }

  renderer.render(scene, camera);
}

// Start the loop
renderer.setAnimationLoop(animate);

Maze Game

  • example.js
import * as THREE from "three";

// 1. Create scene
const scene = new THREE.Scene();
scene.background = new THREE.Color(0xe8e8e8);

// 2. Create camera
const camera = new THREE.PerspectiveCamera(
    55,
    window.innerWidth / window.innerHeight,
    0.1,
    100
);

camera.position.set(0, 12, 10);
camera.lookAt(0, 0, 0);

// 3. Create renderer
const renderer = new THREE.WebGLRenderer();
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

// 4. Add lights
const light = new THREE.DirectionalLight(0xffffff, 2);
light.position.set(-5, 10, 5);
scene.add(light);

scene.add(new THREE.AmbientLight(0xffffff, 0.7));

// 5. Maze design
// # = Wall, . = Path, S = Start, G = Goal

const maze = [
    "#########",
    "#S....#.#",
    "#####.#.#",
    "#...#.#.#",
    "#.#.#.#.#",
    "#.#...#.#",
    "#.#.###.#",
    "#......G#",
    "#########"
];

const cellSize = 1.2;
const offset = (maze.length * cellSize) / 2;

const walls = [];

let startPosition = new THREE.Vector3();
let goalPosition = new THREE.Vector3();

// 6. Create floor
const floor = new THREE.Mesh(
    new THREE.BoxGeometry(maze[0].length * cellSize, 0.2, maze.length * cellSize),
    new THREE.MeshStandardMaterial({ color: 0xcccccc })
);

floor.position.y = -0.1;
scene.add(floor);

// 7. Create maze walls and goal
for (let row = 0; row < maze.length; row++) {

    for (let col = 0; col < maze[row].length; col++) {

        const x = col * cellSize - offset + cellSize / 2;
        const z = row * cellSize - offset + cellSize / 2;

        const cell = maze[row][col];

        if (cell === "#") {

            const wall = new THREE.Mesh(
                new THREE.BoxGeometry(1.2, 0.8, 1.2),
                new THREE.MeshStandardMaterial({ color: 0x555555 })
            );

            wall.position.set(x, 0.4, z);

            scene.add(wall);
            walls.push(wall);
        }

        if (cell === "S") {
            startPosition.set(x, 0.35, z);
        }

        if (cell === "G") {

            goalPosition.set(x, 0.02, z);

            const goal = new THREE.Mesh(
                new THREE.BoxGeometry(0.9, 0.04, 0.9),
                new THREE.MeshStandardMaterial({ color: 0x00ff00 })
            );

            goal.position.copy(goalPosition);
            scene.add(goal);
        }
    }
}

// 8. Create player ball
const ball = new THREE.Mesh(
    new THREE.SphereGeometry(0.32, 24, 24),
    new THREE.MeshStandardMaterial({ color: 0xff0000 })
);

ball.position.copy(startPosition);
scene.add(ball);

// 9. Keyboard controls
const keys = {};

document.addEventListener("keydown", (event) => {
    keys[event.key.toLowerCase()] = true;
});

document.addEventListener("keyup", (event) => {
    keys[event.key.toLowerCase()] = false;
});

// 10. Check wall collision
function hitsWall(position) {

    const ballBox = new THREE.Box3().setFromCenterAndSize(
        position,
        new THREE.Vector3(0.64, 0.64, 0.64)
    );

    for (const wall of walls) {

        const wallBox = new THREE.Box3().setFromObject(wall);

        if (ballBox.intersectsBox(wallBox)) {
            return true;
        }
    }

    return false;
}

// 11. Move ball
let wallHit = false;
let levelComplete = false;

function moveBall() {

    if (levelComplete) return;

    const speed = 0.06;

    const oldPosition = ball.position.clone();

    if (keys["w"] || keys["arrowup"])
        ball.position.z -= speed;

    if (keys["s"] || keys["arrowdown"])
        ball.position.z += speed;

    if (keys["a"] || keys["arrowleft"])
        ball.position.x -= speed;

    if (keys["d"] || keys["arrowright"])
        ball.position.x += speed;

    // Check collision with walls
    if (hitsWall(ball.position)) {

        ball.position.copy(oldPosition);

        if (!wallHit) {
            console.log("Wall Hit");
            wallHit = true;
        }

    } else {
        wallHit = false;
    }

    // Check goal
    if (ball.position.distanceTo(goalPosition) < 0.55) {

        levelComplete = true;

        document.getElementById("message").style.display = "block";
    }
}

// 12. Animation loop
function animate() {

    requestAnimationFrame(animate);

    moveBall();

    renderer.render(scene, camera);
}

animate();

    camera.aspect = window.innerWidth / window.innerHeight;

    camera.updateProjectionMatrix();

    renderer.setSize(window.innerWidth, window.innerHeight);
});
